cmake_minimum_required(VERSION 3.20)
project(softing_core_probe LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "This SDK probe requires the Windows x86 toolchain")
endif()
if(NOT DEFINED SOFTING_SDK_ROOT)
  message(FATAL_ERROR "Set SOFTING_SDK_ROOT to the pinned SDK checkout")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/normalize-sdk-path.cmake")
normalize_sdk_path("${SOFTING_SDK_ROOT}" SOFTING_SDK_ROOT_NORMALIZED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${CMAKE_COMMAND}" -E env "SOFTING_SDK_ROOT=${SOFTING_SDK_ROOT_NORMALIZED}"
  "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../verify-softing-sdk.py"
  RESULT_VARIABLE pin_result)
if(NOT pin_result EQUAL 0)
  message(FATAL_ERROR "Softing SDK pin/header verification failed")
endif()
set(core "${SOFTING_SDK_ROOT_NORMALIZED}/development/core")
file(GLOB core_sources CONFIGURE_DEPENDS
  "${core}/src/SOCmn/*.cpp" "${core}/src/SOClt/*.cpp" "${core}/src/SODaC/*.cpp")
option(OPCL_USE_LEGACY_OPC "Use historical SDK OPC inputs for CI comparison only" OFF)
option(OPCL_BUILD_PROBES "Build test probes and enable SDK test hooks" OFF)
if(OPCL_USE_LEGACY_OPC AND NOT OPCL_BUILD_PROBES)
  message(FATAL_ERROR "Legacy OPC inputs are permitted only for probe/control builds")
endif()
if(OPCL_USE_LEGACY_OPC)
  if(DEFINED OPCF_MIT_STAGED OR DEFINED OPCF_MIT_GENERATED)
    message(FATAL_ERROR "Legacy and official OPC inputs cannot be mixed")
  endif()
  file(GLOB opc_sources CONFIGURE_DEPENDS "${core}/src/opc/*_i.c")
  set(opc_include "${core}/src/opc")
else()
  if(NOT DEFINED OPCF_MIT_STAGED OR NOT DEFINED OPCF_MIT_GENERATED)
    message(FATAL_ERROR "Official OPCF inputs are required; set OPCF_MIT_STAGED and OPCF_MIT_GENERATED")
  endif()
  include("${CMAKE_CURRENT_SOURCE_DIR}/opcf-mit-inputs.cmake")
  opcf_mit_inputs(opc_sources opc_include)
endif()
list(APPEND core_sources ${opc_sources})
# SDK embeds Expat sources into one C++ TU. Build the same pinned C sources
# as separate C translation units so winconfig.h controls its own Windows
# includes and its private types do not collide with the SDK headers.
include("${CMAKE_CURRENT_SOURCE_DIR}/filter-embedded-expat.cmake")
filter_embedded_expat(core_sources)
set(expat "${core}/src/SOCmn/expat")
add_library(softing_expat STATIC "${expat}/xmlrole.c" "${expat}/xmltok.c" "${expat}/xmlparse.c")
target_include_directories(softing_expat PRIVATE "${core}/src/include" "${expat}")
target_compile_definitions(softing_expat PRIVATE XML_STATIC COMPILED_FROM_DSP)
if(OPCL_BUILD_PROBES)
  add_executable(softing_expat_smoke expat-smoke.c)
  target_include_directories(softing_expat_smoke PRIVATE "${core}/src/include")
  target_compile_definitions(softing_expat_smoke PRIVATE XML_STATIC)
  target_link_libraries(softing_expat_smoke PRIVATE softing_expat)
endif()
add_library(softing_da_bridge STATIC ../../../../native/softing-da/adapter.cpp ../../../../native/softing-da/runtime.cpp ${core_sources})
if(OPCL_BUILD_PROBES)
  add_executable(softing_core_probe main.cpp)
  target_link_libraries(softing_core_probe PRIVATE softing_da_bridge)
  target_compile_definitions(softing_da_bridge PUBLIC OPCL_DA_TESTING)
endif()
target_include_directories(softing_da_bridge PUBLIC
  "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../../../../native/softing-da" "${core}/src/include" "${opc_include}"
  "${core}/src/common" "${core}/common")
target_compile_definitions(softing_da_bridge PUBLIC
  _WINDOWS WIN32 _UNICODE UNICODE _WIN32_DCOM _CRT_SECURE_NO_WARNINGS _CRT_NON_CONFORMING_SWPRINTFS)
target_compile_options(softing_da_bridge PUBLIC /W3 /Zc:wchar_t- /EHsc)
target_link_libraries(softing_da_bridge PUBLIC softing_expat ole32 oleaut32 uuid advapi32 ws2_32 mpr rpcrt4)

if(NOT OPCL_USE_LEGACY_OPC)
  add_dependencies(softing_da_bridge opcf_mit_guard)
  target_compile_options(softing_da_bridge PRIVATE /showIncludes)
  if(OPCL_BUILD_PROBES)
    add_dependencies(softing_core_probe opcf_mit_guard)
    target_compile_options(softing_core_probe PRIVATE /showIncludes)
    target_link_options(softing_core_probe PRIVATE /VERBOSE:LIB)
  endif()
endif()
